$rol_nombre = "Usuario";

// Stats
$res_total = mysqli_query($conexion, "SELECT COUNT(*) as c FROM equipo");

// Ultimos movimientos
$sql_ultimos = "SELECT h.fecha_asignacion, e.nombre_equipo, CONCAT(u.nombre, ' ', u.apellido) as user
                FROM historial_equipo h
                JOIN equipo e ON h.id_equipo = e.id_equipo
                JOIN usuario u ON h.id_usuario = u.id_usuario
                ORDER BY h.fecha_asignacion DESC LIMIT 4";
$res_ultimos = mysqli_query($conexion, $sql_ultimos);
